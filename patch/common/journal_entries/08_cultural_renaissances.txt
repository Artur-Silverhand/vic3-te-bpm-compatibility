je_ukraine_cultural_renaissance = {
	icon = "gfx/interface/icons/event_icons/waving_flag.dds"
	group = je_group_global_international_situations
	dlc = dlc019
	is_shown_when_inactive = {
		tbcp_cr_ukraine_has_homelands = yes
		has_dlc_feature = ip5_content
		cu:ukrainian = {
			culture_current_fervor >= 5
		}
	}
	activation_description = {
		always = yes
	}
	possible = {
		tbcp_cr_ukraine_has_homelands = yes
		OR = {
			cu:ukrainian = {
				any_scope_homeland_state = {
					owner = {
						is_involved_in_journal_entry = je_springtime_of_the_peoples
					}
				}
			}
			cu:ukrainian = {
				culture_current_fervor >= 15
			}
		}
	}
	should_be_involved = {
		any_scope_state = {
			state_region = {
				is_homeland = cu:ukrainian
			}
		}
	}
	should_show_when_not_involved = {
		culture_percent_country = {
			target = cu:ukrainian
			value > 0.01
		}
	}
	immediate = {
		set_global_variable = {
			name = tbcp_cr_ukraine_active_days
			value = 0
		}
		set_global_variable = {
			name = tbcp_cr_ukraine_last_day
			value = game_date
		}
	}
	immediate_all_involved = {
		tbcp_cr_ukraine_join = yes
	}
	reason = {
		first_valid = {
			triggered_desc = {
				desc = je_ukraine_cultural_renaissance_reason
				trigger = {
					NOT = {
						has_variable = hromada_var
					}
					NOT = {
						has_variable = prosvita_var
					}
				}
			}
			triggered_desc = {
				desc = je_ukraine_cultural_renaissance_reason_hromada_permitted
				trigger = {
					var:hromada_var ?= 1
				}
			}
			triggered_desc = {
				desc = je_ukraine_cultural_renaissance_reason_hromada_surveilled
				trigger = {
					var:hromada_var ?= 2
				}
			}
			triggered_desc = {
				desc = je_ukraine_cultural_renaissance_reason_hromada_suppressed
				trigger = {
					var:hromada_var ?= 3
				}
			}
			triggered_desc = {
				desc = je_ukraine_cultural_renaissance_reason_prosvita_state_funded
				trigger = {
					var:prosvita_var ?= 1
				}
			}
			triggered_desc = {
				desc = je_ukraine_cultural_renaissance_reason_prosvita_self_funded
				trigger = {
					var:prosvita_var ?= 2
				}
			}
			triggered_desc = {
				desc = je_ukraine_cultural_renaissance_reason_prosvita_suppressed
				trigger = {
					var:prosvita_var ?= 3
				}
			}
		}
	}
	scripted_progress_bar = je_cultural_renaissance_ukraine_progress_bar
	on_monthly_pulse = {
		effect = {
			tbcp_cr_ukraine_reconcile = yes
			if = {
				limit = {
					tbcp_cr_ukraine_has_homelands = yes
				}
				scope:journal_entry = {
					every_scope_je_involved = {
						limit = {
							tbcp_cr_ukraine_country_has_homeland = yes
						}
						if = {
							limit = {
								is_player = no
								NOR = {
									has_variable = ai_ukraine_renaissance_repress
									has_variable = ai_ukraine_renaissance_moderate
									has_variable = ai_ukraine_renaissance_accelerate
								}
							}
							cultural_renaissance_ai_approach_roll = {
								NAME = ukraine
								CULTURE = ukrainian
							}
						}
						if = {
							limit = {
								has_variable = hromada_var
							}
							every_scope_state = {
								limit = {
									state_region = {
										is_homeland = cu:ukrainian
									}
									NOT = {
										has_modifier = hromada
									}
								}
								random = {
									chance = 1
									if = {
										limit = {
											prev.var:hromada_var ?= 1
										}
										add_modifier = {
											name = hromada
											days = long_modifier_time
										}
										post_notification = hromada_appears_permitted
									}
									else_if = {
										limit = {
											prev.var:hromada_var ?= 2
										}
										add_modifier = {
											name = hromada
											days = normal_modifier_time
											multiplier = 0.5
										}
										post_notification = hromada_appears_surveilled
									}
									else_if = {
										limit = {
											prev.var:hromada_var ?= 3
										}
										add_modifier = {
											name = hromada
											days = short_modifier_time
											multiplier = 0.5
										}
										post_notification = hromada_appears_suppressed
									}
								}
							}
						}
						if = {
							limit = {
								has_variable = prosvita_var
							}
							every_scope_state = {
								limit = {
									state_region = {
										is_homeland = cu:ukrainian
									}
									NOR = {
										has_modifier = government_funded_prosvita
										has_modifier = self_funded_prosvita
									}
								}
								random = {
									chance = 1
									if = {
										limit = {
											prev.var:prosvita_var ?= 1
										}
										add_modifier = {
											name = government_funded_prosvita
											days = long_modifier_time
										}
										post_notification = prosvita_appears_state_funded
									}
									else_if = {
										limit = {
											prev.var:prosvita_var ?= 2
										}
										add_modifier = {
											name = self_funded_prosvita
											days = normal_modifier_time
										}
										post_notification = prosvita_appears_self_funded
									}
									else_if = {
										limit = {
											prev.var:prosvita_var ?= 3
										}
										add_radicals_in_state = {
											value = small_radicals
											culture = cu:ukrainian
										}
										post_notification = prosvita_appears_suppressed
									}
								}
							}
						}
						random_list = {
							100 = {

							}
							10 = {
								trigger_event = cultural_renaissances_events.1
							}
							10 = {
								trigger_event = cultural_renaissances_events.2
							}
							10 = {
								trigger_event = cultural_renaissances_events.3
							}
							10 = {
								trigger_event = cultural_renaissances_events.4
							}
							10 = {
								trigger_event = cultural_renaissances_events.5
							}
						}
					}
				}
			}
		}
	}
	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			tbcp_cr_ukraine_has_homelands = yes
			custom_tooltip = {
				text = je_cultural_renaissance_bar_max
				scope:journal_entry = {
					scripted_bar_progress = {
						name = je_cultural_renaissance_ukraine_progress_bar
						value >= 100
					}
				}
			}
		}
		effect_all_involved = {
			if = {
				limit = {
					tbcp_cr_ukraine_country_has_homeland = yes
				}
				if = {
					limit = {
						cu:ukrainian = {
							culture_has_national_awakening = no
						}
					}
					start_national_awakening = {
						culture = cu:ukrainian
						months = 240
					}
				}
				random_political_movement = {
					limit = {
						is_political_movement_type = movement_cultural_minority
						culture = cu:ukrainian
					}
					hidden_effect = {
						remove_modifier = cultural_renaissance
					}
					custom_tooltip_no_bullet = ukraine_national_movement
					add_modifier = {
						name = cultural_renaissance
						days = very_long_modifier_time
						multiplier = 2
					}
				}
				custom_tooltip_no_bullet = je_cultural_renaissance_ukraine_independent_modifier
				show_as_tooltip = {
					add_modifier = {
						name = cultural_foundation
						days = very_long_modifier_time
					}
				}
				custom_tooltip = je_cultural_renaissance_ukraine_event_independence
				set_global_variable = {
					name = cultural_renaissance_ukraine_var
					value = 100
				}
				hidden_effect = {
					remove_variable = hromada_var
					remove_variable = prosvita_var
					if = {
						limit = {
							has_variable = ai_ukraine_renaissance_repress
						}
						remove_variable = ai_ukraine_renaissance_repress
					}
					if = {
						limit = {
							has_variable = ai_ukraine_renaissance_moderate
						}
						remove_variable = ai_ukraine_renaissance_moderate
					}
					if = {
						limit = {
							has_variable = ai_ukraine_renaissance_accelerate
						}
						remove_variable = ai_ukraine_renaissance_accelerate
					}
				}
			}
		}
	}
	outcome = {
		header = je_standard_failure_header

					}
				}
			}
		}
	}

		conclusion_desc = JOURNAL_ENTRY_TIMEOUT_CONCLUSION_DESC

					}

					}
				}
				custom_tooltip = modifier_scaled_by_progress_bar
				hidden_effect = {
					set_global_variable = {
						name = cultural_renaissance_ukraine_var
						value = "scope:journal_entry.scripted_bar_progress(je_cultural_renaissance_ukraine_progress_bar)"
					}
					remove_variable = hromada_var
					remove_variable = prosvita_var
					if = {
						limit = {
							has_variable = ai_ukraine_renaissance_repress
						}
						remove_variable = ai_ukraine_renaissance_repress
					}
					if = {
						limit = {
							has_variable = ai_ukraine_renaissance_moderate
						}
						remove_variable = ai_ukraine_renaissance_moderate
					}
					if = {
						limit = {
							has_variable = ai_ukraine_renaissance_accelerate
						}
						remove_variable = ai_ukraine_renaissance_accelerate
					}
				}
			}
		}
		trigger = {
			tbcp_cr_ukraine_has_homelands = yes
			global_var:tbcp_cr_ukraine_active_days >= 7300
		}
	}
	map_color = {
		if = {
			limit = {
				"culture_percent_state(cu:ukrainian)" >= 0.05
			}
			lerp = {
				min_color = "ukraine_palest_blue"
				max_color = "ukraine_blue"
				factor = "culture_percent_state(cu:ukrainian)"
			}
		}
	}
	map_color_second = {
		if = {
			limit = {
				OR = {
					has_modifier = hromada
					has_modifier = government_funded_prosvita
					has_modifier = self_funded_prosvita
				}
			}
			value = "ukraine_yellow"
		}
	}
	map_tooltip = {
		owner ?= {
			custom_tooltip_no_bullet = "je_ukraine_cultural_renaissance_map_fervor"
		}
		if = {
			limit = {
				"culture_percent_state(cu:ukrainian)" >= 0.05
			}
			save_temporary_scope_value_as = {
				name = ukrainian_culture_percent
				value = "culture_percent_state(cu:ukrainian)"
			}
			custom_tooltip_no_bullet = "je_ukraine_cultural_renaissance_map_culture_percent"
		}
		if = {
			limit = {
				state_region = {
					is_homeland = cu:ukrainian
				}
			}
			custom_tooltip_no_bullet = "je_ukraine_cultural_renaissance_map_homeland"
		}
		if = {
			limit = {
				has_modifier = hromada
			}
			custom_tooltip_no_bullet = je_ukraine_cultural_renaissance_map_hromada
		}
		if = {
			limit = {
				OR = {
					has_modifier = government_funded_prosvita
					has_modifier = self_funded_prosvita
				}
			}
			custom_tooltip_no_bullet = je_ukraine_cultural_renaissance_map_prosvita
		}
	}
	weight = 100
	should_be_pinned_by_default_uninvolved_or_context = no
	should_be_pinned_by_default_involved = yes
	on_become_involved_after_activation = {
		tbcp_cr_ukraine_join = yes
	}
	progress_desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					NOT = {
						tbcp_cr_ukraine_has_homelands = yes
					}
				}
				desc = tbcp_cr_paused
			}
			triggered_desc = {
				trigger = {
					always = yes
				}
				desc = tbcp_cr_ukraine_clock
			}
		}
	}
