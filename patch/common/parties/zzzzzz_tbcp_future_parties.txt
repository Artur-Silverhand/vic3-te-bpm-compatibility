		if = { # transhumanists are natural allies
			limit = {
				leader ?= {
					has_ideology = ideology:ideology_optimist_transhumanist
				}
			}
			add = {
				value = 8
				desc = "from_transhumanist"
			}
		}
		if = { # market liberals favor technocratic efficiency
			limit = {
				leader ?= {
					has_ideology = ideology:ideology_market_liberal
				}
			}
			add = {
				value = 5
				desc = "from_market_liberal"
			}
		}
		if = { # moderates appreciate evidence-based policy
			limit = {
				leader ?= {
					has_ideology = ideology:ideology_moderate
				}
			}
			add = {
				value = 5
				desc = "from_moderate"
			}
		}
		if = { # reformers drawn to rational governance
			limit = {
				leader ?= {
					OR = {
						has_ideology = ideology:ideology_reformer
						has_ideology = ideology:ideology_liberal_leader
					}
				}
			}
			add = {
				value = 3
				desc = "from_reformer"
			}
		}
		if = { # industrialists are the core base
			limit = {
				is_interest_group_type = ig_industrialists
			}
			add = {
				value = 5
				desc = "from_industrialist_base"
			}
		}
		if = { # intelligentsia provides expertise
			limit = {
				is_interest_group_type = ig_intelligentsia
			}
			add = {
				value = 4
				desc = "from_expertise"
			}
		}
		if = { # environmentalists can align if technocratic approaches help
			limit = {
				leader ?= {
					has_ideology = ideology:ideology_environmentalists
				}
			}
			add = {
				value = 2
				desc = "from_green_tech"
			}
		}
		if = { # reactionaries reject technocracy
			limit = {
				leader ?= {
					has_ideology = ideology:ideology_reactionary
				}
			}
			add = {
				value = -8
				desc = "from_reactionary"
			}
		}
		if = { # populists are anti-expert
			limit = {
				leader ?= {
					has_ideology = ideology:ideology_ethno_nationalist
				}
			}
			add = {
				value = -8
				desc = "from_populist_opposition"
			}
		}
		if = { # devout distrust secular technocrats
			limit = {
				is_interest_group_type = ig_devout
			}
			add = {
				value = -5
				desc = "from_religious_skepticism"
			}
		}
		if = { # rural folk see technocrats as distant elitists
			limit = {
				is_interest_group_type = ig_rural_folk
			}
			add = {
				value = -4
				desc = "from_rural_skepticism"
			}
		}
		if = { # socialists and communists oppose corporate technocracy
			limit = {
				leader ?= {
					OR = {
						has_ideology = ideology:ideology_communist_leader
						has_ideology = ideology:ideology_vanguardist
						has_ideology = ideology:ideology_state_socialist
					}
				}
			}
			add = {
				value = -5
				desc = "from_socialist_opposition"
			}
		}
		if = { # landowners resist disruption of traditional hierarchies
			limit = {
				is_interest_group_type = ig_landowners
			}
			add = {
				value = -3
				desc = "from_landowner_tradition"
			}
		}
		if = { # jingoists don't care for technocratic restraint
			limit = {
				leader ?= {
					OR = {
						has_ideology = ideology:ideology_jingoist
						has_ideology = ideology:ideology_jingoist_leader
					}
				}
			}
			add = {
				value = -3
				desc = "from_militarist"
			}
		}
		# First Past the Post (amendment_first_past_the_post): one of the two
		# largest parties, as te_fptp_effects.txt ranks them monthly. In place
		# of the party-size multipliers (the else_ifs below), so a major party
		# never fills up: each group joining first takes a little off instead.
		if = {
			limit = {
				owner = {
					te_fptp_in_force = yes
					has_variable = te_fptp_two_majors
					has_variable = te_fptp_major_technocratic_party
				}
			}
			add = {
				value = te_fptp_major_party_join_weight
				desc = "te_fptp_major_party_join"
			}
			if = {
				limit = { scope:number > 0 }
				add = {
					value = scope:number
					multiply = te_fptp_major_party_crowding_weight
					desc = "te_fptp_major_party_crowding"
				}
			}
		}
		else_if = {
			limit = {
				scope:number = 1
				is_powerful = no
				ig_counts_as_marginal = no
			}
			multiply = {
				value = 2
				desc = "from_influential_supporting_member"
			}
		}
		else_if = {
			limit = {
				scope:number = 2
			}
			multiply = {
				value = 0.5
				desc = "from_two_prior_members"
			}
		}
		else_if = {
			limit = {
				scope:number = 3
			}
			multiply = {
				value = 0.05
				desc = "from_three_prior_members"
			}
		}
		else_if = {
			limit = {
				scope:number > 3
			}
			multiply = {
				value = 0.0
				desc = "from_more_than_three_prior_members"
			}
		}
		# First Past the Post: any party but the two largest.
		if = {
			limit = {
				owner = {
					te_fptp_in_force = yes
					has_variable = te_fptp_two_majors
					NOT = { has_variable = te_fptp_major_technocratic_party }
				}
			}
			add = {
				value = te_fptp_minor_party_join_weight
				desc = "te_fptp_minor_party_join"
			}
		}
	}
}
