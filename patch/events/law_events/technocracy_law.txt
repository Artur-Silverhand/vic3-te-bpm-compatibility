technocracy.4 = {
	type = country_event

	title = technocracy.4.t
	desc = technocracy.4.d
	flavor = technocracy.4.f

	event_image = {
		video = "unspecific_gears_pistons"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_industry.dds"

	duration = 3

	cooldown = { days = normal_modifier_time }

	trigger = {
		# Triggered by JE completion
	}

	immediate = {
		ig:ig_intelligentsia ?= {
			save_scope_as = intelligentsia_ig
		}
	}

	option = {
		name = technocracy.4.a
		default_option = yes
		ai_chance = {
			base = 1
			modifier = {
				trigger = {
					ai_avoids_radicals = yes
				}
				add = 10
			}
		}
		add_modifier = {
			name = modifier_technocracy_brave_new_world
			days = normal_modifier_time
		}
		scope:intelligentsia_ig = {
			if = {
				limit = {
					has_ideology = ideology:ideology_republican
				}
				remove_ideology = ideology_republican
				add_ideology = ideology_technocratic
			}
			else_if = {
				limit = {
					has_ideology = ideology:ideology_constitutionalist
				}
				remove_ideology = ideology_constitutionalist
				add_ideology = ideology_technocratic
			}
		}
	}

	option = {
		name = technocracy.4.b
		add_modifier = {
			name = modifier_technocracy_scientific_management
			days = long_modifier_time
		}
		scope:intelligentsia_ig = {
			if = {
				limit = {
					has_ideology = ideology:ideology_republican
				}
				remove_ideology = ideology_republican
				add_ideology = ideology_technocratic
			}
			else_if = {
				limit = {
					has_ideology = ideology:ideology_constitutionalist
				}
				remove_ideology = ideology_constitutionalist
				add_ideology = ideology_technocratic
			}
		}
	}

	after = {
		je_technocracy_cleanup = yes
	}
}
