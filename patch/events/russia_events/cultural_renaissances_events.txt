cultural_renaissances_events.1 = {
	type = country_event
	placement = scope:src_state
	duration = 3
	title = cultural_renaissances_events.1.t
	desc = cultural_renaissances_events.1.d
	flavor = cultural_renaissances_events.1.f
	event_image = {
		video = "ip5_theater"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/waving_flag.dds"
	dlc = dlc019
	fire_only_once = yes
	trigger = {
		tbcp_cr_ukraine_country_has_homeland = yes
		AND = {
			has_state_in_state_region = STATE_EAST_GALICIA
			s:STATE_EAST_GALICIA = {
				is_homeland = cu:ukrainian
			}
		}
		NOT = {
			c:UKR ?= this
		}
	}
	immediate = {
		s:STATE_EAST_GALICIA = {
			random_scope_state = {
				limit = {
					owner = root
				}
				save_scope_as = src_state
			}
		}
	}
	option = {
		name = cultural_renaissances_events.1.a
		ai_chance = {
			base = 10
			modifier = {
				trigger = {
					has_variable = ai_ukraine_renaissance_accelerate
				}
				add = 100
			}
		}
		if = {
			limit = {
				tbcp_cr_ukraine_country_has_homeland = yes
			}
			add_modifier = {
				name = supreme_ruthenian_council
				days = long_modifier_time
			}
			cu:ukrainian = {
				add_fervor = 10
			}
			if = {
				limit = {
					exists = c:HUN
					NOT = {
						c:HUN ?= this
					}
				}
				change_relations = {
					country = c:HUN
					value = -5
				}
			}
			else_if = {
				limit = {
					AND = {
						has_state_in_state_region = STATE_RUTHENIA
						s:STATE_RUTHENIA = {
							is_homeland = cu:ukrainian
						}
					}
				}
				random_scope_state = {
					limit = {
						state_region = s:STATE_RUTHENIA
					}
					add_radicals_in_state = {
						value = small_radicals
						culture = cu:hungarian
					}
				}
			}
		}
	}
	option = {
		name = cultural_renaissances_events.1.bb
		ai_chance = {
			base = 10
			modifier = {
				trigger = {
					has_variable = ai_ukraine_renaissance_moderate
				}
				add = 100
			}
		}
		if = {
			limit = {
				tbcp_cr_ukraine_country_has_homeland = yes
			}
			add_modifier = {
				name = ruthenian_representatives
				days = long_modifier_time
			}
			cu:ukrainian = {
				add_fervor = 5
			}
		}
	}
	option = {
		name = cultural_renaissances_events.1.c
		default_option = yes
		ai_chance = {
			base = 10
			modifier = {
				trigger = {
					has_variable = ai_ukraine_renaissance_repress
				}
				add = 100
			}
		}
