# Completion event 2, choose ideology
communism.2 = {
	type = country_event
	placement = root
	title = communism.2.t
	desc = communism.2.d
	flavor = communism.2.f

	duration = 3

	event_image = {
		video = "ip5_soviet_deliberation"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_newspaper.dds"

	trigger = {
		# triggered by communsim JE 2
	}

	fire_only_once = yes

	immediate = {
		ig:ig_trade_unions ?= {
			save_scope_as = unions
			leader = {
				save_scope_as = union_leader
				set_temporary_event_immortality = { DURATION = 3 }
			}
		}
	}

	option = { # communism
		name = communism.2.b
		default_option = yes
		ig:ig_trade_unions ?= {
			set_variable = chose_communism
		}
		custom_tooltip = chose_communism.tt
		if = {
			limit = {
				scope:union_leader = {
					NOT = { has_ideology = ideology:ideology_communist_leader }
				}
			}
			scope:union_leader = {
				set_ideology = ideology:ideology_communist_leader
			}
		}
		ai_chance = {
			base = 50
		}
	}
	option = { # vanguardism
		name = communism.2.a
		trigger = {
			has_technology_researched = political_agitation
			custom_tooltip = {
				text = vanguardism_viable_tt
				vanguardism_coefficient > 1
			}
		}
		show_as_unavailable = {
			has_technology_researched = political_agitation
		}
		highlighted_option = yes
		ig:ig_trade_unions ?= {
			set_variable = chose_vanguardism
		}
		custom_tooltip = chose_vanguardism.tt
		if = {
			limit = {
				scope:union_leader = {
					NOT = { has_ideology = ideology:ideology_vanguardist }
				}
			}
			scope:union_leader = {
				set_ideology = ideology:ideology_vanguardist
			}
		}
		ai_chance = {
			base = 10
			modifier = {
				trigger = {
					vanguardism_coefficient >= 2
				}
				add = 40
			}
		}
	}
	option = { # anarchism
		name = communism.2.c
		trigger = {
			any_political_movement = {
				is_political_movement_type = movement_anarchist
			}
		}
		show_as_unavailable = {
			has_technology_researched = anarchism
		}
		highlighted_option = yes
		ig:ig_trade_unions ?= {
			set_variable = chose_anarchism
		}
		custom_tooltip = chose_anarchism.tt
		if = {
			limit = {
				scope:union_leader = {
					NOT = { has_ideology = ideology:ideology_anarchist }
				}
			}
			scope:union_leader = {
				set_ideology = ideology:ideology_anarchist
			}
		}
		ai_chance = {
			base = 0
			modifier = {
				trigger = {
					any_political_movement = {
						is_political_movement_type = movement_anarchist
						political_movement_support >= 0.1
					}
				}
				add = 25
			}
		}
	}

	after = {
		revoke_temporary_event_immortality = { CHARACTER = scope:union_leader }
		if = {
			limit = {
				NOT = { has_global_variable = socialist_world_republic }
			}
			add_journal_entry = {
				type = je_state_and_revolution
			}
		}
		add_journal_entry = {
			type = je_social_revolution
		}
		custom_tooltip = {
			text = form_white_movement_tt
			if = {
				limit = {
					NOT = {
						any_political_movement = {
							is_political_movement_type = movement_anti_communist
						}
					}
				}
				create_political_movement = {
					type = movement_anti_communist
				}
			}
		}
		je_communism_2_cleanup = yes
	}
}

communism.3 = {
	type = country_event
	placement = root
	title = communism.3.t
	desc = communism.3.d
	flavor = communism.3.f

	duration = 3

	event_image = {
		video = "votp_agitator_speaking"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_newspaper.dds"

	cooldown = { days = short_modifier_time }

	trigger = {
		# triggered by journal entry - je_extraordinary_commission [on_weekly_pulse]
		has_technology_researched = political_agitation
		any_interest_group = {
			ig_counts_as_marginal = yes
			OR = {
				is_interest_group_type = ig_rural_folk
				is_interest_group_type = ig_intelligentsia
				is_interest_group_type = ig_armed_forces
				is_interest_group_type = ig_trade_unions
			}
			any_scope_character = {
				NOR = {
					has_role_of_type = agitator
					has_role_of_type = ruler
				}
				has_ideology = ideology:ideology_vanguardist
			}
			leader = {
				NOR = {
					has_ideology = ideology:ideology_vanguardist
					has_ideology = ideology:ideology_communist_leader
					has_ideology = ideology:ideology_anarchist
				}
			}
		}
	}

	immediate = {
		random_interest_group = {
			limit = {
				ig_counts_as_marginal = yes
				OR = {
					is_interest_group_type = ig_rural_folk
					is_interest_group_type = ig_intelligentsia
					is_interest_group_type = ig_armed_forces
					is_interest_group_type = ig_trade_unions
				}
				any_scope_character = {
					NOR = {
						has_role_of_type = agitator
						has_role_of_type = ruler
					}
					has_ideology = ideology:ideology_vanguardist
				}
				leader = {
					NOR = {
						has_ideology = ideology:ideology_vanguardist
						has_ideology = ideology:ideology_communist_leader
						has_ideology = ideology:ideology_anarchist
					}
				}
			}
			save_scope_as = new_vanguard_ig
			set_variable = chose_vanguardism
			ordered_scope_character = {
				limit = {
					NOR = {
						has_role_of_type = agitator
						has_role_of_type = ruler
					}
					has_ideology = ideology:ideology_vanguardist
				}
				order_by = prominence
				position = 0
				save_scope_as = new_vanguard_leader
				set_temporary_event_immortality = { DURATION = 3 }
				set_as_interest_group_leader = yes
			}
		}
	}

	cancellation_trigger = {
		NOT = {
			has_journal_entry = je_extraordinary_commission
		}
	}

	after = {
		revoke_temporary_event_immortality = { CHARACTER = scope:new_vanguard_leader }
	}

	option = {
		name = communism.3.a
		default_option = yes
		custom_tooltip = new_vanguard.tt
		scope:new_vanguard_ig = {
			add_modifier = {
				name = new_vanguard_opinion
				days = normal_modifier_time
			}
		}
		ai_chance = {
			base = 40
		}
	}
	option = {
		name = communism.3.b
		custom_tooltip = new_vanguard.tt
		scope:new_vanguard_ig = {
			add_modifier = {
				name = new_vanguard_clout
				days = normal_modifier_time
			}
		}
		ai_chance = {
			base = 60
		}
	}
}

 		has_law_or_variant = law_type:law_council_republic

			}

			add_radicals_in_state = {
				value = small_radicals

			}
		}
		custom_tooltip = {
			text = communism_var_tick_05_tt
			set_variable = { name = communism_progress_var_add value = 0.5 }
		}
		ai_chance = {
			base = 100
			modifier = {
				trigger = {
					ai_avoids_radicals = yes
				}
				add = 10
			}
		}
	}

	after = {
		if = {
			limit = {
				has_technology_researched = socialism
				country_has_laws_in_need_of_critical_modernization = no
				NOT = {
					any_political_movement = {
						is_political_movement_type = movement_socialist
					}
				}
			}
			create_political_movement = { type = movement_socialist }
		}
		else_if = {
			limit = {
				has_technology_researched = anarchism
				country_has_laws_in_need_of_critical_modernization = no
				NOT = {
					any_political_movement = {
						OR = {
							is_political_movement_type = movement_socialist
							is_political_movement_type = movement_anarchist
						}
					}
				}
			}
			create_political_movement = { type = movement_anarchist }
		}
