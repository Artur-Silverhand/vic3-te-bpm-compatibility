		video = "asia_westerners_arriving"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_fire.dds"

	cooldown = { days = long_modifier_time }

	cancellation_trigger = {

	}

	immediate = {
		ordered_interest_group = {
			limit = {
				is_in_government = yes
			}
			order_by = ig_clout
			position = 0
			check_range_bounds = no
			save_scope_as = ruling_ig
			leader = {
				save_scope_as = tairo_scope
				set_temporary_event_immortality = { DURATION = 3 }
			}
		}
		hidden_effect = {
			set_strategy = ai_strategy_industrial_expansion
			set_strategy = ai_strategy_meiji_restoration
			set_strategy = ai_strategy_maintain_power_balance
		}
	}

	option = {
		name = ep2_meiji.1.a
		default_option = yes
		if = {
			limit = {
				has_modifier = forced_market_opening
			}
			every_scope_character = {
				limit = {
					has_ideology = ideology:ideology_shojoi
				}
				add_modifier = {
					name = shogun_forced_to_open_market_character
					days = very_long_modifier_time
					is_decaying = yes
				}
			}
			random_political_movement = {
				limit = {
					is_political_movement_type = movement_tokugawa_loyalist
				}
				add_modifier = {
					name = shogun_forced_to_open_market
					days = long_modifier_time
					is_decaying = yes
				}
			}
		}
		if = {
			limit = {
				owner = {
					any_political_movement = {
						is_political_movement_type = movement_tokugawa_loyalist
						has_core_ideology = ideology:ideology_tokugawa_loyalist_movement
					}
				}
			}
			owner = {
				random_political_movement = {
					limit = {
						is_political_movement_type = movement_tokugawa_loyalist
						has_core_ideology = ideology:ideology_tokugawa_loyalist_movement
					}
					set_core_ideology = ideology_tokugawa_loyalist_movement_2
					add_character_ideology = ideology_modernizer_leader
				}
			}
		}
	}

	after = {
		revoke_temporary_event_immortality = { CHARACTER = scope:tairo_scope }
	}
}

ep2_meiji_pulse.1 = { # Namamugi Incident
	type = country_event
	placement = scope:relevant_treaty_port_state

	duration = 3

	title = ep2_meiji_pulse.1.t
	desc = ep2_meiji_pulse.1.d
	flavor = ep2_meiji_pulse.1.f

    event_image = {
		video = "ep2_important_japanese_men_negotiating"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_skull.dds"

	fire_only_once = yes

	cancellation_trigger = {
		has_revolution = yes
	}

	trigger = {
		# triggered by journal entry - je_meiji_restoration [on_monthly_pulse]
		any_state_in_home_islands = {
			is_treaty_port = yes
			owner = { NOT = { has_war_with = root } }
			any_neighbouring_state = {
				owner = ROOT
			}
		}
		any_in_global_list = {
			variable = great_power_country_list
			NOT = { ROOT = this }
			has_strategic_region_interest_tier = {
				strategic_region = ROOT.capital.region
				value >= 1
			}
		}
		any_political_movement = {
			is_political_movement_type = movement_meiji_restorationist
			political_movement_radicalism > define:NPolitics|RADICALISM_AGITATING_THRESHOLD
		}
		any_scope_character = {
			has_variable = daimyo_var
			NOT = {
				var:daimyo_var ?= s:STATE_KYOTO # Hikone Domain is landlocked
			}
		}
	}

	immediate = {
		random_political_movement = {
			limit = {
				is_political_movement_type = movement_meiji_restorationist
			}
			save_scope_as = restorationist_movement_scope
		}
		generate_1_common_name = {
			CULTURE_1 = primary_culture
		}
		random_state_in_home_islands = {
			limit = {
				owner = ROOT
				any_neighbouring_state = {
					is_treaty_port = yes
				}
			}
			save_scope_as = relevant_treaty_port_state
		}
		random_in_global_list = {
			variable = great_power_country_list
			limit = {
				NOT = { ROOT = this }
				has_strategic_region_interest_tier = {
					strategic_region = ROOT.capital.region
					value >= 1
				}
			}
			save_scope_as = relevant_gp
			random_primary_culture = {
				save_scope_as = relevant_gp_culture
			}
		}
		scope:relevant_gp ?= {
			create_character = {
				historical = yes
				culture = primary_culture
				role = character_role_magnate
				interest_group = ig_industrialists
				ideology = ideology_moderate
				holding_type = building_financial_district
				traits = {
					arrogant
					explorer
				}
				on_created = {
					set_temporary_event_immortality = { DURATION = 3 }
					transfer_character = c:JAP
				}
				save_scope_as = richardson_scope
			}
		}
		random_scope_character = {
			limit = {
				has_variable = daimyo_var
				NOT = {
					var:daimyo_var ?= s:STATE_KYOTO # Hikone Domain is landlocked
				}
			}
			save_scope_as = daimyo_scope
			set_temporary_event_immortality = { DURATION = 3 }
			set_variable = namamugi_daimyo_var
		}
	}

	option = {
		name = ep2_meiji_pulse.1.a
		default_option = yes
		scope:richardson_scope ?= {
			show_as_tooltip = {
				kill_character = yes
			}
			hidden_effect = {
				place_character_in_void = 100
			}
		}
		scope:relevant_gp ?= {
			trigger_event = { id = ep2_meiji_pulse.11 days = 30 }
		}
	}

	after = {
		clear_generated_names = yes
		revoke_temporary_event_immortality = { CHARACTER = scope:richardson_scope }
		revoke_temporary_event_immortality = { CHARACTER = scope:daimyo_scope }
	}
}
