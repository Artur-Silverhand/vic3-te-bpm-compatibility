BRZ_populism.4 = {
	type = country_event
	placement = scope:coffee_ocean_state

	title = BRZ_populism.4.t
	desc = BRZ_populism.4.d
	flavor = BRZ_populism.4.f

	event_image = {
		video = "europenorthamerica_russian_serfs"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_trade.dds"

	cooldown = { days = normal_modifier_time }

	duration = 3

	dlc = dlc008

	trigger = {
		# triggered by journal entry - je_populist_unrest [on_monthly_pulse]
		OR = {
			has_law_or_variant = law_type:law_tenant_farmers
			has_law_or_variant = law_type:law_commercialized_agriculture
			has_law_or_variant = law_type:law_peasant_proprietorship
		}
		any_scope_state = {
			any_scope_building = {
				is_building_type = building_coffee_plantation
				level >= 15
			}
		}
		market = {
			mg:coffee = {
				market_goods_cheaper >= 0.35
			}
		}
	}

	immediate = {
		random_scope_state = {
			limit = {
				any_scope_building = {
					is_building_type = building_coffee_plantation
					level >= 15
				}
			}
			save_scope_as = coffee_ocean_state
		}
	}

	option = {
		name = BRZ_populism.4.a
		default_option = yes
		add_radicals = {
			value = medium_radicals
			strata = middle
		}
		add_radicals = {
			value = medium_radicals
			strata = lower
		}
		scope:coffee_ocean_state = {
			add_modifier = {
				name = modifier_coffee_ocean_state_poverty
				days = normal_modifier_time
			}
		}
	}

	option = {
		name = BRZ_populism.4.b
		add_modifier = {
			name = modifier_welfare_compromises_brazil
			days = normal_modifier_time
		}
		add_radicals = {
			value = small_radicals
			strata = middle
		}
		add_radicals = {
			value = small_radicals
			strata = lower
		}
	}
}

#Tenetist Revolt
BRZ_populism.6 = {
	type = country_event
	placement = scope:revolt_state

	title = BRZ_populism.6.t
	desc = BRZ_populism.6.d
	flavor = BRZ_populism.6.f

	event_image = {
		video = "unspecific_devastation"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_fire.dds"

	cooldown = { days = normal_modifier_time }

	duration = 3

	dlc = dlc008

	trigger = {
		# triggered by journal entry - je_populist_unrest [on_monthly_pulse]
		any_scope_state = {
			turmoil >= 0.25
			is_incorporated = yes
			NOT = {
				is_capital = yes
			}
		}
	}

	immediate = {
		ordered_scope_state = {
			limit = {
				turmoil >= 0.25
				is_incorporated = yes
				NOT = {
					is_capital = yes
				}
			}
			order_by = gdp
			position = 0
			check_range_bounds = no
			save_scope_as = revolt_state
			set_variable = tenetist_revolt_state
		}
	}

	option = {
		name = BRZ_populism.6.a
		default_option = yes
		if = {
			limit = {
				NOT = {
					has_variable = tenetist_revolt_happened
				}
			}
			set_variable = tenetist_revolt_happened
		}
		scope:revolt_state = {
			add_radicals_in_state = {
				value = very_large_radicals
				strata = middle
			}
			add_radicals_in_state = {
				value = medium_radicals
				strata = lower
			}
			state_region = {
				add_devastation = 10
			}
		}
	}

	option = {
		name = BRZ_populism.6.b
		trigger = {
			any_military_formation = {
				has_mobilization_option = mobilization_option:mobilization_option_aerial_recon
			}
		}
		if = {
			limit = {
				NOT = {
					has_variable = tenetist_revolt_happened
				}
			}
			set_variable = tenetist_revolt_happened
		}
		scope:revolt_state = {
			add_radicals_in_state = {
				value = medium_radicals
				strata = middle
			}
			add_radicals_in_state = {
				value = small_radicals
				strata = lower
			}
			state_region = {
				add_devastation = 20
			}
		}
	}
}
