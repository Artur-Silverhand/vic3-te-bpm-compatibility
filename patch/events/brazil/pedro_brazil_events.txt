# Landowners or AF are Marginalized
pedro.5 = {
	type = country_event
	placement = ROOT

	title = pedro.5.t
	desc = pedro.5.d
	flavor = {
		first_valid = {
			triggered_desc = {
				desc = pedro.5.f
				trigger = {
					scope:marginal_ig_scope = {
						is_interest_group_type = ig_armed_forces
					}
				}
			}
			triggered_desc = {
				desc = pedro.5.f2
				trigger = {
					always = yes
				}
			}
		}
	}

	event_image = {
		video = "southamerica_aristocrats"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_default.dds"

	duration = 3

	dlc = dlc008

	fire_only_once = yes

	trigger = {
		# triggered by journal entry - je_pedro_brazil [on_monthly_pulse]
		# triggered by je_pedro_brazil
		OR = {
			AND = {
				NOT = { has_variable = pedro_lo_marginal_var }
				ig:ig_landowners ?= {
					ig_counts_as_marginal = yes
				}
			}
			AND = {
				NOT = { has_variable = pedro_af_marginal_var }
				ig:ig_armed_forces ?= {
					ig_counts_as_marginal = yes
				}
			}

		}
	}

	immediate = {
		if = {
			limit = {
				NOT = { has_variable = pedro_lo_marginal_var }
				ig:ig_landowners ?= {
					ig_counts_as_marginal = yes
				}
			}
			set_variable = pedro_lo_marginal_var
			ig:ig_landowners ?= {
				save_scope_as = marginal_ig_scope
			}
		}
		else = {
			set_variable = pedro_af_marginal_var
			ig:ig_armed_forces ?= {
				save_scope_as = marginal_ig_scope
			}
		}
		ruler = {
			save_scope_as = pedro_scope
			set_temporary_event_immortality = { DURATION = 3 }
		}
	}

	after = {
		revoke_temporary_event_immortality = { CHARACTER = scope:pedro_scope }
	}

	option = {
		name = pedro.5.a
		default_option = yes
		add_pedropoint_effect = yes
	}
}

# Enactment event
pedro.11 = {
	type = country_event
	placement = ROOT

	title = pedro.11.t
	desc = pedro.11.d
	flavor = pedro.11.f

	event_image = {
		video = "unspecific_signed_contract"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_fire.dds"

	duration = 3

	dlc = dlc008

	category = enactment

	cooldown = { days = long_modifier_time }

	trigger = {
		# triggered by journal entry - je_pedro_brazil [on_monthly_pulse]
		has_journal_entry = je_pedro_republic
		is_enacting_law = law_type:law_presidential_republic
	}

	cancellation_trigger = {
		OR = {
			ruler = {
				NOT = {
					has_variable = is_pedro
				}
			}
			NOT = {
				exists = root.currently_enacting_law
			}
		}
	}

	immediate = {
		ruler = {
			save_scope_as = pedro_scope
			set_temporary_event_immortality = { DURATION = 3 }
		}
		currently_enacting_law = {
			save_scope_as = current_law_scope
		}
	}

	after = {
		revoke_temporary_event_immortality = { CHARACTER = scope:pedro_scope }
	}

	option = {
		name = pedro.11.a
		default_option = yes
		add_enactment_phase = 1
		add_loyalists = {
			value = 0.05
		}
	}
}

			remove_modifier = isabel_heir_popularity
		}
		scope:pedro_scope = {
			remove_modifier = isabel_heir_popularity
		}
		add_pedropoint_effect_3 = yes
	}
